---
title: Basic Agent
description: A one-agent workflow.
---

```wire
input {
    message: string
}

secrets {
    api_key: string
}

provider llm from openai {
    endpoint: "https://api.openai.com/v1"
    api_key: secrets.api_key
}

model fast from llm {
    id: "gpt-4.1-mini"
}

agent reply {
    model: model.fast
    instruction: "Reply to {{ input.message }}."

    output: {
        message: string
    }
}

output {
    message: agent.reply.message
}
```
