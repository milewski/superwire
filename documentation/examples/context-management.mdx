---
title: Context Management
description: Pass structured context between workflow stages.
---

```wire
secrets {
    endpoint: string
    api_key: string
    models: {
        flash: string
        pro: string
        max: string
    }
}

provider ai {
    driver: "openai"
    endpoint: secrets.endpoint
    api_key: secrets.api_key
    models: {
        flash: secrets.models.flash
        pro: secrets.models.pro
        max: secrets.models.max
    }
}

input {
    notes: string
}

agent extract_context {
    model: ai("flash")
    instruction: "Extract facts, constraints, and open questions from these notes: {{ input.notes }}."
    output: {
        facts: [string]
        constraints: [string]
        questions: [string]
    }
}

agent write_summary {
    model: ai("flash")
    context: {
        facts: agent.extract_context.facts
        constraints: agent.extract_context.constraints
    }
    instruction: "Write a summary using the provided context."
    output: {
        summary: string
    }
}

output {
    summary: agent.write_summary.summary
    open_questions: agent.extract_context.questions
}
```

Use `context` for structured values and keep `instruction` focused on the task.
