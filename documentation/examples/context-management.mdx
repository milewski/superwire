---
title: Context Management
description: Express full-context and compacted-context flows in the current DSL.
---

This example is based on `crates/core/workflows/context_compaction.ai`.

```ai-dsl
provider openai {
    driver: "openai"
    endpoint: "https://api.openai.com/v1"
    api_key: "test-api-key"
    models: ["gpt-4.1-mini"]
}

input {
    product_name: string
    customer_feedback: [string]
}

agent research_analysis {
    model: openai("gpt-4.1-mini")
    prompt: "Analyze this customer feedback for {{ input.product_name }}: {{ input.customer_feedback }}"
    output: {
        themes: [string]
        risks: [string]
        opportunities: [string]
    }
}

agent summary_from_full_context {
    model: openai("gpt-4.1-mini")
    context: context(agent.research_analysis)
    prompt: "Write a short executive summary from the available context."
    output: string
}

agent summary_from_custom_compact_context {
    model: openai("gpt-4.1-mini")
    context: compact(
        agent: agent.research_analysis,
        model: openai("gpt-4.1-mini"),
        inference: {
            temperature: 0.1
            max_tokens: 1_500
        },
        prompt: "Keep only the highest-signal themes and risks."
    )
    prompt: "Write a short executive summary from the customized compacted context."
    output: {
        summary: string
        top_actions: [string; 3]
    }
}
```

`context(...)` and `compact(...)` execute in the current runtime and can be used in `context:` and `output` expressions.
