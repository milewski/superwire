---
title: MCP Tools
description: Fetch context from an MCP tool and use it in an agent.
---

```wire
secrets {
    endpoint: string
    api_key: string
    mcp_summarizer_endpoint: string
    models: {
        flash: string
        pro: string
        max: string
    }
}


provider ai {
    driver: "openai"
    endpoint: secrets.endpoint
    api_key: secrets.api_key
    models: {
        flash: secrets.models.flash
        pro: secrets.models.pro
        max: secrets.models.max
    }
}

input {
    project_id: number
    task_id: number
}

mcp summarizer {
    endpoint: secrets.mcp_summarizer_endpoint
}

tool fetch_task from mcp.summarizer.tool.fetch_task {
    bindings {
        project_id: input.project_id
    }
    input {
        task_id: number
    }
    output {
        title: string
        description: string
    }
}

dynamic {
    task: call tool.fetch_task {
        input {
            task_id: input.task_id
        }
    }
}

agent summarize_task {
    model: ai("flash")
    context: {
        task: dynamic.task
    }
    instruction: "Summarize the task context and suggest the next action."
    output: {
        summary: string
        next_action: string
    }
}

output {
    task: dynamic.task
    summary: agent.summarize_task.summary
    next_action: agent.summarize_task.next_action
}
```

This pattern is deterministic: the workflow always fetches task context before the agent runs.
