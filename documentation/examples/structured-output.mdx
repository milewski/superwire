---
title: Structured Output
description: Return typed fields from an agent.
---

```wire
input {
    message: string
}

secrets {
    api_key: string
}

provider llm from openai {
    endpoint: "https://api.openai.com/v1"
    api_key: secrets.api_key
}

model fast from llm {
    id: "gpt-4.1-mini"
}

agent classify {
    model: model.fast
    instruction: "Classify {{ input.message }}."

    output: {
        category: enum { bug, feature, question }
        confidence: number
        reason: string
    }
}

output {
    result: agent.classify
}
```
