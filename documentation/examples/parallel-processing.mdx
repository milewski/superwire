---
title: Parallel Processing
description: Run independent agents and loop iterations concurrently.
---

```wire
secrets {
    endpoint: string
    api_key: string
    models: {
        flash: string
        pro: string
        max: string
    }
}

provider ai {
    driver: "openai"
    endpoint: secrets.endpoint
    api_key: secrets.api_key
    models: {
        flash: secrets.models.flash
        pro: secrets.models.pro
        max: secrets.models.max
    }
}

input {
    topic: string
    task_ids: [number]
}

agent risks {
    model: ai("flash")
    instruction: "List risks for {{ input.topic }}."
    output: {
        risks: [string]
    }
}

agent opportunities {
    model: ai("flash")
    instruction: "List opportunities for {{ input.topic }}."
    output: {
        opportunities: [string]
    }
}

agent summarize_each for task_id in input.task_ids {
    model: ai("flash")
    instruction: "Summarize task {{ task_id }}."
    output: {
        task_id: number
        summary: string
    }
}

agent final_plan {
    model: ai("pro")
    instruction: "Create a plan from risks {{ agent.risks.risks }}, opportunities {{ agent.opportunities.opportunities }}, and task summaries {{ agent.summarize_each }}."
    output: {
        plan: string
    }
}

output {
    risks: agent.risks.risks
    opportunities: agent.opportunities.opportunities
    summaries: agent.summarize_each
    plan: agent.final_plan.plan
}
```

The two top-level analysis agents and each loop iteration can run before `final_plan`.
