---
title: Multi-Agent Workflow
description: Chain agents through structured references.
---

```wire
secrets {
    endpoint: string
    api_key: string
    models: {
        flash: string
        pro: string
        max: string
    }
}

provider ai {
    driver: "openai"
    endpoint: secrets.endpoint
    api_key: secrets.api_key
    models: {
        flash: secrets.models.flash
        pro: secrets.models.pro
        max: secrets.models.max
    }
}

input {
    topic: string
}

agent collect_facts {
    model: ai("flash")
    instruction: "List facts about {{ input.topic }}."
    output: {
        facts: [string]
    }
}

agent write_brief {
    model: ai("pro")
    instruction: "Write a concise brief using these facts: {{ agent.collect_facts.facts }}."
    output: {
        title: string
        brief: string
    }
}

output {
    title: agent.write_brief.title
    brief: agent.write_brief.brief
    facts: agent.collect_facts.facts
}
```

`write_brief` runs after `collect_facts` because it references `agent.collect_facts.facts`.
