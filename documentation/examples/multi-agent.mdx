---
title: Multi-Agent Workflow
description: Chain agents through structured references.
---

```wire
input {
    topic: string
}

secrets {
    api_key: string
}

provider llm from openai {
    endpoint: "https://api.openai.com/v1"
    api_key: secrets.api_key
}

model fast from llm {
    id: "gpt-4.1-mini"
}

agent research {
    model: model.fast
    instruction: "List important facts about {{ input.topic }}."

    output: {
        facts: [string]
    }
}

agent write_summary {
    model: model.fast
    instruction: "Write a summary using these facts: {{ agent.research.facts }}."

    output: {
        summary: string
    }
}

output {
    facts: agent.research.facts
    summary: agent.write_summary.summary
}
```
