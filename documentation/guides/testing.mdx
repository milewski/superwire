title: Testing
description: Parse, validate, and execute workflows inside Rust tests.

Superwire provides several ways to test workflows without a live LLM provider.

## Parse and validate without execution

The fastest tests only parse and validate:

```rust
use superwire_core::dsl::{parse_workflow, validate_workflow};

#[test]
fn valid_workflow_parses() {
    let source = r#"
        provider openai from openai {
endpoint: "https://api.openai.com/v1"
}

model openai_model from openai {
    id: "gpt-4.1-mini"
}

        agent greeting {
            model: model.openai_model
            instruction: "Say hello."
            output: string
        }

        output {
            greeting: agent.greeting
        }
    "#;

    let workflow = parse_workflow(source).expect("parse failed");
    let report = validate_workflow(&workflow);
    assert!(report.is_valid(), "validation failed: {:?}", report.issues);
}
```

## Detect validation errors

```rust
use superwire_core::dsl::{parse_workflow, validate_workflow};

#[test]
fn secret_leak_detected() {
    let source = r#"
        secrets {
            api_key: string
        }

        provider openai from openai {
endpoint: "https://api.openai.com/v1"
            api_key: secrets.api_key
}

model openai_model from openai {
    id: "gpt-4.1-mini"
}

        agent greeting {
            model: model.openai_model
            instruction: "The secret is {{ secrets.api_key }}"
            output: string
        }

        output {
            greeting: agent.greeting
        }
    "#;

    let workflow = parse_workflow(source).expect("parse failed");
    let report = validate_workflow(&workflow);
    assert!(!report.is_valid());
}
```

## Execute with a mock provider

For full execution tests without external dependencies, implement a custom `AgentRunner`:

```rust
use superwire_core::runtime::*;
use superwire_core::dsl::parse_workflow;
use std::sync::Arc;

struct MockRunner;

#[async_trait::async_trait]
impl AgentRunner for MockRunner {
    async fn run_agent(
        &self,
        _request: AgentExecutionRequest,
    ) -> Result<AgentExecutionResult, WorkflowRuntimeError> {
        Ok(AgentExecutionResult {
            output: serde_json::json!("mock response"),
            context: serde_json::Value::Null,
        })
    }
}

#[tokio::test]
async fn execute_with_mock() {
    let source = r#"
        provider mock from openai {
endpoint: "http://localhost:1234"
}

model mock_model from mock {
    id: "model-a"
}

        agent greeting {
            model: model.mock_model
            instruction: "Say hello."
            output: string
        }

        output {
            greeting: agent.greeting
        }
    "#;

    let workflow = parse_workflow(source).unwrap();
    let runtime = WorkflowRuntime::new(Arc::new(MockRunner));
    let output = runtime.execute(&workflow, None).await.unwrap();

    assert_eq!(output["greeting"], "mock response");
}
```

## Testing fixture files

The repository includes `.wire` fixture files in `crates/executor/tests/fixtures/`. You can use these in your own tests:

```rust
use std::fs;

#[test]
fn fixture_parses() {
    let source = fs::read_to_string(
        "crates/executor/tests/fixtures/001_minimum.wire"
    ).unwrap();
    let workflow = superwire_core::dsl::parse_workflow(&source);
    assert!(workflow.is_ok());
}
```

## Testing dynamic blocks and MCP integrations

For workflows that use `dynamic` blocks with `call tool.xxx`, provide a custom runner that handles the registered tools:

```rust
use superwire_core::runtime::*;

struct ToolAwareRunner;

#[async_trait::async_trait]
impl AgentRunner for ToolAwareRunner {
    async fn run_agent(
        &self,
        request: AgentExecutionRequest,
    ) -> Result<AgentExecutionResult, WorkflowRuntimeError> {
        // Access tools from request.tools
        // Execute them as needed and return results
        Ok(AgentExecutionResult {
            output: serde_json::json!({"status": "ok"}),
            context: serde_json::Value::Null,
        })
    }
}
```

See the [Rust API](/api-reference/rust-api) page for the full runtime interface.
