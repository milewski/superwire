```wire loop.wire
input {
    tasks: [{ id: number, title: string, description: string }]
}

secrets {
    api_key: string
}

provider llm from openai {
    api_key: secrets.api_key
}

model fast from llm {
    id: "gpt-4.1-mini"
}

agent analyze for { id, title, description } in input.tasks {
    model: model.fast
    instruction: "Analyze task {{ id }}: {{ title }}. Details: {{ description }}"

    output {
        task_id: number
        summary: string
        risk: enum { low, medium, high }
    }
}

output {
    analyses: agent.analyze
}
```
